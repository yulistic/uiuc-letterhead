% https://tex.stackexchange.com/questions/59932/reproducing-an-official-letterhead
%
% Header and footer follow the official stationery sample
% (https://brand.illinois.edu/applying-the-brand/stationery): the unit block
% in the header is Source Sans (unit name bold, address regular, UIblue); the
% footer's "UNIVERSITY OF ILLINOIS URBANA-CHAMPAIGN" is Montserrat bold caps
% in UIorange and the contact line below it is Source Sans in UIblue. Both
% typefaces are in TeX Live (montserrat, sourcesanspro).
%
% The body is set in Source Sans, the brand's body typeface. For a serif body
% replace the "Body font" block below with, e.g., \usepackage{newtxtext}.

%% Colors
% https://www.creativeservices.illinois.edu/brand/logos-and-colors.html
% blue:   CMYK 100/90/10/77
% orange: CMYK 01/83/85/0
\usepackage[rgb]{xcolor}
\definecolor{UIblue}{RGB}{19,41,75}
\definecolor{UIorange}{RGB}{232,74,39}

% set page geometry
% top: the header block ends at about 1.65in; bottom: the footer block starts
% at 9.94in from the top, so the body must stop above 9.8in.
\usepackage[top=1.95in,left=1.0in,bottom=1.2in,right=1.0in]{geometry}
\usepackage{graphicx}

% logo
\providecommand\logofile{Illinois-Wordmark-Horizontal-Full-Color-CMYK.pdf}

% remove coloring from URLs
\usepackage[colorlinks=false,pdfborder={0 0 0}]{hyperref}
\urlstyle{same}

% to place things in exact places
%\usepackage[absolute,showboxes]{textpos}
\usepackage[absolute]{textpos}

% for conditionals
\usepackage{ifthen}

\usepackage[T1]{fontenc}

% Body font
\usepackage[default,lining]{sourcesanspro}

% Letterhead typefaces, selected by family name in the header and footer so
% that they do not depend on the body font.
\usepackage{montserrat}
\newcommand*{\lhMontserrat}{\fontfamily{Montserrat-TLF}\selectfont}
\newcommand*{\lhSourceSans}{\fontfamily{SourceSansPro-TLF}\selectfont}

% set linespacing and paragraph spacing
\linespread{1.10}
\setlength{\parskip}{1em}

% use microtype to extend spacing
\usepackage{microtype}

% Remove paragraph indentation
\parindent0pt
\raggedright%

% change page style to include header/footer
\pagestyle{empty}

% add stuff to the doc
\AtBeginDocument{%
    % Top block with logo
    \begin{textblock*}{\textwidth}(1.0in,0.75in)
      \includegraphics[height=0.3125in]{\logofile}
    \end{textblock*}

    % Inset department address, right-aligned to the body's right margin
    % (paper width 8.5in - right margin 1.0in = 7.5in)
    \begin{textblock*}{4.0in} (3.5in,0.45in)
      \begin{flushright}
        \lhSourceSans\color{UIblue}%
        {\fontsize{10}{12}\fontseries{b}\selectfont \fromcollege\par}
        \vspace{1mm}
        {\fontsize{9}{11}\selectfont \fromdept\\\fromdeptaddress\par}
      \end{flushright}
    \end{textblock*}
  }

\usepackage{fancyhdr}
\pagestyle{fancy}

\renewcommand{\footrulewidth}{0pt}
\fancyfoot{}
\fancyfoot[L]{%
    % footer
    % specify the top right corner location of the footer
    \begin{textblock*}{\textwidth}[1.0,0.0](7.5in,9.9375in)
        \begin{flushleft}
        {\lhMontserrat\fontsize{9.5}{12}\fontseries{b}\selectfont\color{UIorange}%
         UNIVERSITY OF ILLINOIS URBANA-CHAMPAIGN\par}
        \vspace{0.5mm}
        {\lhSourceSans\fontsize{9}{11}\selectfont\color{UIblue}%
         \fromtel\ \textbullet\ \fromweb\par}
        \end{flushleft}
    \end{textblock*}
    }

\fancyhead{}
\renewcommand{\headrulewidth}{0pt}

% add closing to end of the document, if sig is available
\AtEndDocument{%
    \par\vspace{2ex}
    \fromname%
}
